/* * IoT Smart Energy Meter - ECE 3rd Year Project
 * Features:Voltage, Current, Power, Energy, Cost Estimation & Blynk Analytics
 */

// --- BLYNK CREDENTIALS ---
#define BLYNK_TEMPLATE_ID "TMPL3vrfyH7I7"
#define BLYNK_TEMPLATE_NAME "Energy Meter"
#define BLYNK_AUTH_TOKEN "YOUR_BLYNK_AUTH_TOKEN"

#include <ESP8266WiFi.h>
#include <BlynkSimpleEsp8266.h>
#include <Wire.h>
#include <Adafruit_ADS1X15.h>
#include <LiquidCrystal_I2C.h>

// --- WIFI CREDENTIALS ---
char ssid[] = "YOUR_WIFI_NAME";
char pass[] = "YOUR_PASSWORD";

Adafruit_ADS1115 ads;
LiquidCrystal_I2C lcd(0x27, 16, 2);
BlynkTimer timer;

// --- CALIBRATION, SAFETY & BILLING ---
int vOffset = 13300, iOffset = 13300; 
float vCal = 675.0, iCal = 25.0;  
float currentLimit = 5.0;      // Safety Limit in Amps
float unitPrice = 7.0;         // Set your cost per unit (e.g., 7.0 INR)
bool alertSent = false;

// --- FILTER & DATA VARIABLES ---
float smoothV = 0, smoothI = 0, alpha = 0.02;
float finalV = 0, finalI = 0, Power = 0;
float totalCost = 0;
unsigned long lastMillis = 0;
double energyWh = 0;

void checkSafety() {
  if (finalI > currentLimit) {
    if (!alertSent) {
      Blynk.logEvent("overload_alert", String("Overload! Current: ") + finalI + "A");
      alertSent = true;
    }
  } else {
    alertSent = false;
  }
}

void sendDataToBlynk() {
  // 1. Calculate Billing
  float totalUnits = energyWh / 1000.0;
  totalCost = totalUnits * unitPrice;

  // 2. Push to Blynk Cloud
  Blynk.virtualWrite(V0, finalV);      // Voltage
  Blynk.virtualWrite(V1, finalI);      // Current
  Blynk.virtualWrite(V2, Power);       // Power (For Graphs)
  Blynk.virtualWrite(V3, totalUnits);  // Energy (For Graphs)
  Blynk.virtualWrite(V4, totalCost);   // Cost Estimation
  
  checkSafety();
}

void setup() {
  Serial.begin(115200);
  lcd.init();
  lcd.backlight();
  lcd.print("Smart IoT Meter");
  
  if (!ads.begin()) {
    lcd.setCursor(0, 1);
    lcd.print("ADS Error!");
    while (1);
  }

  Blynk.begin(BLYNK_AUTH_TOKEN, ssid, pass);
  timer.setInterval(2000L, sendDataToBlynk);
  lastMillis = millis();
}

void loop() {
  Blynk.run();
  timer.run();

  int16_t vMax = 0, iMax = 0;
  unsigned long start = millis();

  // Capture peak over 100ms
  while (millis() - start < 100) { 
    int16_t vRead = abs(ads.readADC_SingleEnded(0) - vOffset);
    int16_t iRead = abs(ads.readADC_SingleEnded(1) - iOffset);
    if (vRead > vMax) vMax = vRead;
    if (iRead > iMax) iMax = iRead;
    yield(); 
  }

  // Convert and Smooth
  float instantV = (vMax * 0.1875 / 1000.0) * vCal;
  float instantI = (iMax * 0.1875 / 1000.0) * iCal;
  smoothV = (smoothV * (1.0 - alpha)) + (instantV * alpha);
  smoothI = (smoothI * (1.0 - alpha)) + (instantI * alpha);

  finalV = (smoothV < 95.0) ? 0 : smoothV; 
  finalI = (smoothI < 0.20) ? 0 : smoothI;
  Power = (finalV > 0 && finalI > 0) ? (finalV * finalI) : 0;

  // Energy Tracking
  unsigned long currentMillis = millis();
  energyWh += (Power * (currentMillis - lastMillis)) / 3600000.0;
  lastMillis = currentMillis;

  // Local LCD Display (Cycles every 2.5 seconds to show all data)
  static unsigned long lcdTimer = 0;
  if (millis() - lcdTimer > 2500) {
    static bool toggle = false;
    lcd.clear();
    if (!toggle) {
      lcd.print("V:"); lcd.print(finalV, 1);
      lcd.setCursor(9, 0); lcd.print("I:"); lcd.print(finalI, 2);
      lcd.setCursor(0, 1); lcd.print("P:"); lcd.print(Power, 1); lcd.print("W");
    } else {
      lcd.print("Units:"); lcd.print((energyWh/1000.0), 4);
      lcd.setCursor(0, 1); lcd.print("Bill:"); lcd.print(totalCost, 2);
    }
    toggle = !toggle;
    lcdTimer = millis();
  }
}
